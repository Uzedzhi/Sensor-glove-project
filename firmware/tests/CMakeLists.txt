cmake_minimum_required(VERSION 3.16)
project(bno055_host_tests LANGUAGES CXX)
enable_testing()
add_executable(bno055_tests bno055_tests.cpp)
target_include_directories(bno055_tests PRIVATE stubs)
target_compile_features(bno055_tests PRIVATE cxx_std_17)
set_target_properties(bno055_tests PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
target_compile_options(bno055_tests PRIVATE -Wall -Wextra -Werror)
set(BNO055_VIEWER_INCLUDE_DIR "" CACHE PATH "Optional bno055_viz/include directory")
if(BNO055_VIEWER_INCLUDE_DIR)
    if(NOT EXISTS "${BNO055_VIEWER_INCLUDE_DIR}/BnoSample.h")
        message(FATAL_ERROR "BNO055_VIEWER_INCLUDE_DIR must contain BnoSample.h")
    endif()
    target_include_directories(bno055_tests PRIVATE "${BNO055_VIEWER_INCLUDE_DIR}")
    target_compile_definitions(bno055_tests PRIVATE HAVE_BNO055_VIEWER=1)
endif()
add_test(NAME bno055_host COMMAND bno055_tests)
