cmake_minimum_required(VERSION 3.16)
project(glove_mouse LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(BUILD_GUI "Build the raylib configuration window" ON)
find_package(PkgConfig REQUIRED)
pkg_check_modules(JSONC REQUIRED IMPORTED_TARGET json-c)
add_library(glove-core src/json.cpp src/config.cpp src/motion.cpp src/telemetry.cpp src/io.cpp src/runtime.cpp src/ipc.cpp)
target_include_directories(glove-core PUBLIC include)
target_link_libraries(glove-core PUBLIC PkgConfig::JSONC)
target_compile_options(glove-core PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(glove-mouse-driver src/driver_main.cpp)
target_link_libraries(glove-mouse-driver PRIVATE glove-core)
target_compile_options(glove-mouse-driver PRIVATE -Wall -Wextra -Wpedantic -Werror)
if(BUILD_GUI)
    pkg_check_modules(RAYLIB REQUIRED IMPORTED_TARGET raylib)
    add_executable(glove-mouse-config src/gui_main.cpp)
    target_link_libraries(glove-mouse-config PRIVATE glove-core PkgConfig::RAYLIB)
    target_compile_options(glove-mouse-config PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
include(CTest)
if(BUILD_TESTING)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    add_executable(glove-core-tests tests/core_tests.cpp)
    target_link_libraries(glove-core-tests PRIVATE glove-core)
    target_compile_options(glove-core-tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
    add_test(NAME glove-core COMMAND glove-core-tests)
    add_test(NAME glove-driver-integration COMMAND ${Python3_EXECUTABLE}
        ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration.py $<TARGET_FILE:glove-mouse-driver>)
    set_tests_properties(glove-driver-integration PROPERTIES TIMEOUT 40)
endif()
